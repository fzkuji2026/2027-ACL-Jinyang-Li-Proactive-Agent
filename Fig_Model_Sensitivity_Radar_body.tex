\newcommand{\radarseries}[3]{%
  \foreach \v [count=\i] in {#3} {\coordinate (p\i) at ({90-72*(\i-1)}:{\radarR*\v/100});}
  \draw[#1,line join=round] (p1)--(p2)--(p3)--(p4)--(p5)--cycle;
  \foreach \i in {1,...,5} {\node[#2] at (p\i) {};}
}
