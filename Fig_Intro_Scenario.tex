\newcommand{\walkbadge}[3]{% x, y, colour
  \fill[#3] (#1,#2) circle (.24);
  \begin{scope}[shift={(#1,#2)},line cap=round,line join=round,white]
    \fill (.035,-.145) circle (.045);

% user avatar
\newcommand{\avatar}[3]{% x, y, colour

  \end{scope}}

\newcommand{\agent}[3]{% x, y, colour

    \fill[#3,rounded corners=3.5pt] (-.29,-.25) rectangle (.29,.17);
    \fill[white,rounded corners=2.5pt] (-.20,-.15) rectangle (.20,.08);
    \fill[#3] (-.085,-.055) circle (.042) (.085,-.055) circle (.042);
    \draw[#3,line width=.8pt,line cap=round] (-.055,.01) .. controls (-.02,.045) and (.02,.045) .. (.055,.01);
  \end{scope}}
\newcommand{\magnifier}[3]{% x, y, colour
  \draw[#3,line width=1.1pt,fill=white] (#1,#2) circle (.085);
  \draw[#3,line width=1.4pt,line cap=round] (#1+.06,#2+.06)--(#1+.14,#2+.14);}
\newcommand{\tick}[3]{\draw[#3,line width=1pt,line cap=round,line join=round]
  (#1-.06,#2)--(#1-.015,#2+.045)--(#1+.065,#2-.055);}
\newcommand{\cross}[3]{\draw[#3,line width=1pt,line cap=round]
  (#1-.05,#2-.05)--(#1+.05,#2+.05) (#1-.05,#2+.05)--(#1+.05,#2-.05);}
% flat speech bubble with a tail pointing left: x0, y0, x1, y1, colour
\newcommand{\bubble}[5]{%
  \fill[#5] (#1+.02,{(#2+#4)/2-.10})--(#1-.17,{(#2+#4)/2})--(#1+.02,{(#2+#4)/2+.10})--cycle;
  \fill[#5,rounded corners=4pt] (#1,#2) rectangle (#3,#4);}
% recent-event strip ending in the current event
\newcommand{\recent}[1]{% y-centre
  \fill[white,rounded corners=2.5pt] (.30,#1-.19) rectangle (3.35,#1+.19);
  \node[sm,anchor=west] at (.42,#1) {Recent events (72\,h)};
  \foreach \s in {0,...,3}{\fill[blue!30,rounded corners=.7pt] (2.33+\s*.19,#1-.065) rectangle ++(.13,.13);}
  \fill[amber,rounded corners=.8pt] (3.09,#1-.085) rectangle ++(.17,.17);}
% one layer of the memory cylinder: y-top, y-bottom, colour, label
\newcommand{\layer}[4]{%
  \fill[#3] (.42,#1) rectangle (1.62,#2);
  \fill[#3] (1.02,#2) ellipse (.60 and .11);
  \draw[white,line width=.7pt] (.42,#1) arc[start angle=180,end angle=360,x radius=.60,y radius=.11];
  \node[txt,font=\bfseries\fontsize{6.4}{7.6}\selectfont,text=white,anchor=center]
    at (1.02,{(#1+#2)/2+.05}) {#4};}
