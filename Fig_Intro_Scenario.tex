\newcommand{\cmark}[3]{% x, y, colour

\newcommand{\xmark}[3]{%

% Bell = INTERVENE, open circle = ABSTAIN; colour and badge show agreement

\newcommand{\bellicon}[3]{%
  \begin{scope}[shift={(#1,#2)}]
    \fill[#3] (-.15,.10) .. controls (-.15,-.20) and (-.10,-.21) .. (0,-.21)
      .. controls (.10,-.21) and (.15,-.20) .. (.15,.10) -- cycle;
    \fill[#3,rounded corners=.6pt] (-.20,.07) rectangle (.20,.13);
    \fill[#3] (0,.18) circle (.045);
  \end{scope}}
\newcommand{\abstainicon}[3]{%
  \draw[#3,line width=1.1pt,fill=white] (#1,#2) circle (.17);
  \draw[#3,line width=1.1pt,line cap=round] (#1-.08,#2)--(#1+.08,#2);}
\newcommand{\badge}[4]{% x, y, colour, 1=check 0=cross
  \fill[white,draw=#3,line width=.8pt] (#1,#2) circle (.13);
  \ifnum#4=1 \cmark{#1}{#2}{#3}\else\xmark{#1}{#2}{#3}\fi}

    \fill[muted!45] (#1+\f*#2,7.94) circle (.05);

  \fill[amber] (#1+.80*#2,7.94) circle (.065);
  \node[secondary,text=amber,font=\bfseries\fontsize{9.5}{11}\selectfont,
    anchor=center] at (#1+.80*#2,7.67) {18:30};}
